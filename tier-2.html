02 · Lambda Source

Production PII scrubber

Drop this into your vault-scrubber.js handler. Ships ready to deploy.

vault-scrubber.js 
/**
 * Efficiency Labs · Tier 2 — vault-scrubber.js
 * PII sanitization Lambda. Deploys to AWS Lambda / GCP Cloud Functions / Azure Functions.
 * Runs at the edge of every inbound event BEFORE any model call.
 */
import {SecretsManagerClient, GetSecretValueCommand } from "@aws-sdk/client-secrets-manager";

const PII_PATTERNS = [
  { name: "ssn", re: /\b\d{3}-\d{2}-\d{4}\b/g, replacement: "[REDACTED:SSN]" },
  { name: "credit_card",   re: /\b(?:\d[ -]*?){13,16}\b/g,         replacement: "[REDACTED:CC]" },
  { name: "email",        re: /[\w.+-]+@[\w-]+\.[\w.-]+/g, replacement: "[REDACTED:EMAIL]" },
  { name: "phone_us",     re: /\b\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}\b/g, replacement: "[REDACTED:PHONE]" },
  { name: "ipv4",        re: /\b(?:25[0-5]|2[0-4]\d|[01]?\d?\d)(?:\.(?:25[0-5]|2[0-4]\d|[01]?\d?\d)){3}\b/g, replacement: "[REDACTED:IPV4]" },
  { name: "iban",        re: /\b[A-Z]{2}\d{2}[A-Z0-9]{4,30}\b/g, replacement: "[REDACTED:IBAN]" },
  { name: "passport",    re: /\b[A-Z][0-9]{8}\b/g,               replacement: "[REDACTED:PPT]" },
  { name: "dob",         re: /\b(0?[1-9]|1[0-2])\/(0?[1-9]|[12]\d|3[01])\/(19|20)\d{2}\b/g, replacement: "[REDACTED:DOB]" },
  // … 10 more patterns in the actual file
];

export const handler = async (event) => {
 const scrubbed = { ...event, scrubbed_at: new Date().toISOString() };
  const report = { redactions: {}, total_redacted: 0 };

  const walk = (obj) => {
 for (const key in obj) {
      if (typeof obj[key] ==="string") {
        for (const { name, re, replacement } of PII_PATTERNS) {
          const matches = obj[key].match(re);
 if (matches) {
            obj[key] = obj[key].replace(re, replacement);
 report.redactions[name] = (report.redactions[name] || 0) + matches.length;
            report.total_redacted += matches.length;
          }
        }
      } else if (typeof obj[key] === "object" && obj[key] !== null) {
        walk(obj[key]);
      }
    }
  };
  walk(scrubbed);

  // Forward to The Vault Keeper (schema enforce) → LLM agent
  const res = await fetch(process.env.VAULT_KEEPER_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ payload: scrubbed, report })
  });
  return { statusCode: res.status, body: await res.text() };
};
