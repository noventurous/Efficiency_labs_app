runtime/composer.js 
/**
 * Efficiency Labs · Tier 3 — composer.js
 * The OpenMAIC message router. Routes operator messages to the right agent(s),
 * enforces Guardian RBAC, and renders the conversation.
 */
import { mesh }from "./mesh.js";
import { guardian } from "./../guardian/guardian.js";
import { audit } from "./audit.js";

const ROUTING = [
  { match: /diagnostic|readiness|assess/i, agents: ["diagnostician"], tier: 1 },
  { match: /architect|prompt|schema/i,      agents: ["architect"],     tier: 1 },
  { match: /vault|payload|pii|scrub/i,     agents: ["vault"],       tier: 1 },
  { match: /block|isolate|remedi|guardduty/i, agents: ["operator"], tier: 3 },
  { match: /stripe|charge|jira|ticket/i,    agents: ["operator"],    tier: 2 },
  { match: /board|brief|report|audit/i,      agents: ["scribe"],     tier: 1 },
];

export async function route(operatorMessage) {
  const pattern = ROUTING.find(r => r.match.test(operatorMessage)) || ROUTING[0];

  const assessment = await guardian.evaluate({
    message: operatorMessage,
    proposed_agents: pattern.agents,
    proposed_tier: pattern.tier
  });

  await audit.append({ event: "route.match", pattern: pattern.match.source, tier: pattern.tier });

  if (assessment.tier >= 3) {
    return await haltForApproval(operatorMessage, pattern, assessment);
  }

  const results = [];
 for (const agentId of pattern.agents) {
    const out = await mesh.dispatch(agentId, operatorMessage);
 results.push(out);
    await audit.append({ event: "agent.execute", agent: agentId, result: out.summary });
  }
  return { tier: pattern.tier, halted: false, results };
}

async function haltForApproval(msg, pattern, assessment) {
  // Wait for operator 1-click approve/reject in the UI banner
  const decision = await new Promise(resolve => {
 window.openmaicAwaitApproval = resolve;
  });
  await audit.append({ event: "halt.decision", decision, tier: assessment.tier });
  if (!decision)return { tier: assessment.tier, halted: true, approved: false };
  return await route(msg); // retry at reduced tier if approved
}
